	"options": "actual options only", "setups": "installation singleton",

type migrationRecorder struct {
	logger.Interface
	ddl []string
}

func (r *migrationRecorder) Trace(_ context.Context, _ time.Time, fc func() (string, int64), _ error) {
	query, _ := fc()
	upper := strings.ToUpper(strings.TrimSpace(query))
	if strings.HasPrefix(upper, "CREATE ") || strings.HasPrefix(upper, "ALTER ") || strings.HasPrefix(upper, "DROP ") {
		r.ddl = append(r.ddl, query)
	}
}

func emit(value any) {
	b, err := common.Marshal(value)
	if err != nil {
		panic(err)
	}
	fmt.Println(string(b))
}

func main() {
	if os.Getenv("LARGE_DATA_ISOLATED") != "yes" {
		panic("isolated fixture runner required")
	}
	mode := os.Getenv("LARGE_DATA_MODE")
	common.InitEnv()
	if mode == "seed" {
		if err := model.InitDB(); err != nil {
			panic(err)
		}
		if err := model.InitLogDB(); err != nil {
			panic(err)
		}
	} else {
		var dial gorm.Dialector
		dsn := os.Getenv("SQL_DSN")
		switch {
		case strings.HasPrefix(dsn, "postgres://"):
			dial = postgres.Open(dsn)
		case dsn == "":
			dial = sqlite.Open(common.SQLitePath)
		default:
			dial = mysql.Open(dsn)
		}
		db, err := gorm.Open(dial, &gorm.Config{Logger: logger.Default.LogMode(logger.Silent)})
		if err != nil {
			panic(err)
		}
		model.DB, model.LOG_DB = db, db
	}
	db := model.DB.Session(&gorm.Session{Logger: logger.Default.LogMode(logger.Silent), SkipDefaultTransaction: true})
	defer model.CloseDB()
	if mode == "probe" {
		probe(db)
		return
	}
	if mode == "seed-new" || mode == "seed-security" {
		seedNewSchema(db)
	}
	tables, err := db.Migrator().GetTables()
	if err != nil {
		panic(err)
	}
	sort.Strings(tables)
	rows, _ := strconv.Atoi(os.Getenv("LARGE_DATA_ROWS"))
	logRows, _ := strconv.Atoi(os.Getenv("LARGE_DATA_LOG_ROWS"))
	for _, table := range tables {
		if table == "sqlite_sequence" {
			continue
		}
		var count int64
		if err := db.Table(table).Count(&count).Error; err != nil {
			panic(err)
		}
		reason := smallTables[table]
		target := rows
		if table == "logs" || table == "audit_logs" || table == "abilities" {
			target = logRows
		}
		if mode == "seed-new" && !strings.HasPrefix(table, "audit_security_") && table != "channel_creators" {
			reason = "populated in previous-release phase"
		}
		if mode == "count" || reason != "" || count >= int64(target) {
			emit(map[string]any{"table": table, "rows": count, "reason": reason})
			continue
		}
		columns, err := db.Migrator().ColumnTypes(table)
		if err != nil {
			panic(err)
		}
		start := time.Now()
		for offset := int(count); offset < target; offset += 500 {
			batch := make([]map[string]any, 0, 500)
			for j := offset; j < min(offset+500, target); j++ {
				i := j + 1
				row := make(map[string]any, len(columns))
				for _, col := range columns {
					name, typ := col.Name(), strings.ToLower(col.DatabaseTypeName())
					nullable, _ := col.Nullable()
					var value any
					switch {
					case name == "deleted_at":
						if i%10 == 0 {
							value = time.Now().Add(-24 * time.Hour)
						}
					case strings.Contains(typ, "json"):
						value = "{}"
					case strings.Contains(typ, "bool"):
						value = false
					case strings.Contains(typ, "int"), strings.Contains(typ, "numeric"), strings.Contains(typ, "decimal"), strings.Contains(typ, "float"), strings.Contains(typ, "double"), strings.Contains(typ, "real"):
						value = int64(0)
						if name == "id" || strings.HasSuffix(name, "_id") {
							value = int64(i)
						}
						if strings.HasSuffix(name, "_at") || strings.HasSuffix(name, "_time") || name == "at" || name == "last_seen" || name == "first_seen" {
							value = time.Now().Unix() - int64(i%30)*86400 - int64(i%86400)
						}
						if strings.Contains(name, "expires") || strings.Contains(name, "expired") {
							value = time.Now().Add(365 * 24 * time.Hour).Unix()
						}
					case strings.Contains(typ, "time"), strings.Contains(typ, "date"):
						if !nullable {
							value = time.Now().Add(-24 * time.Hour)
						}
					case strings.Contains(typ, "blob"), strings.Contains(typ, "bytea"):
						value = []byte("fixture")
					default:
						text := fmt.Sprintf("f%08d", i)
						if length, ok := col.Length(); ok && length > 0 && int64(len(text)) > length {
							text = text[len(text)-int(length):]
						}
						value = text
					}
					row[name] = value
				}
				fixtureValues(table, i, rows, row)
				batch = append(batch, row)
			}
			if err := db.Table(table).CreateInBatches(batch, 500).Error; err != nil {
				panic(fmt.Sprintf("seed %s at %d: %v", table, offset, err))
			}
		}
		emit(map[string]any{"table": table, "rows": target, "seed_seconds": time.Since(start).Seconds()})
		// Explicit fixture IDs do not advance PostgreSQL sequences. Restore the
		// normal allocator before startup inserts any backfill or system rows.
		if db.Dialector.Name() == "postgres" {
			for _, column := range columns {
				if column.Name() != "id" {
					continue
				}
				var sequence sql.NullString
				if err := db.Raw("SELECT pg_get_serial_sequence(?, ?)", table, "id").Scan(&sequence).Error; err != nil {
					panic(err)
				}
				if sequence.Valid {
					var highest int64
					if err := db.Table(table).Select("MAX(id)").Scan(&highest).Error; err != nil {
						panic(err)
					}
					if err := db.Exec("SELECT setval(?, ?, true)", sequence.String, highest).Error; err != nil {
						panic(err)
					}
				}
			}
		}
	}
	if mode == "seed-new" || mode == "seed-security" {
		recorder := &migrationRecorder{Interface: logger.Default.LogMode(logger.Silent)}
		observed := db.Session(&gorm.Session{Logger: recorder})
		for pass := range 2 {
			recorder.ddl = nil
			start := time.Now()
			seedNewSchema(observed)
			emit(map[string]any{"populated_security_migration": pass + 1, "ms": time.Since(start).Milliseconds(), "ddl": recorder.ddl})
			if len(recorder.ddl) != 0 {
				panic("populated security tables issued repeated DDL")
			}
		}
	}
}

func probe(db *gorm.DB) {
	var reads, writes []float64
	errors := 0
	for os.Getenv("LARGE_DATA_STOP") != "" {
		if _, err := os.Stat(os.Getenv("LARGE_DATA_STOP")); err == nil {
			break
		}
		for _, op := range []string{"read", "write"} {
			ctx, cancel := context.WithTimeout(context.Background(), time.Second)
			start := time.Now()
			var err error
			if op == "read" {
				var at int64
				err = db.WithContext(ctx).Raw("SELECT created_at FROM audit_logs WHERE id = ?", 1).Scan(&at).Error
			} else {
				err = db.WithContext(ctx).Exec("UPDATE audit_logs SET status = status WHERE id = ?", 1).Error
			}
			ms := float64(time.Since(start).Microseconds()) / 1000
			cancel()
			if err != nil {
				errors++
			}
			if op == "read" {
				reads = append(reads, ms)
			} else {
				writes = append(writes, ms)
			}
		}
		time.Sleep(20 * time.Millisecond)
	}
	for k, values := range map[string][]float64{"read": reads, "write": writes} {
		sort.Float64s(values)
		if len(values) > 0 {
			emit(map[string]any{"operation": k, "samples": len(values), "p95_ms": values[(len(values)-1)*95/100], "p99_ms": values[(len(values)-1)*99/100], "max_ms": values[len(values)-1], "total_errors": errors})
		}
	}
}
