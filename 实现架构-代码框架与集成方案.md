## 2. 数据流和集成点

### 2.1 实时数据流

```
Auto Test Task (每N分钟)
        │
        ▼
    testAllChannels()
        │
        ├─► model.RecordConsumeLog()  [既有]
        │
        └─► aggregateGroupStatusRealtime()  [新增]
                │
                ├─► SELECT * FROM consume_log WHERE timestamp > now()-1h
                │
                ├─► Calculate Metrics
                │   ├─ Success Rate = success_count / total_count
                │   ├─ Avg Response = avg(response_time)
                │   ├─ P95 Response = percentile(response_time, 95)
                │   └─ Error Count = error_count
                │
                └─► UPDATE group_status SET ...
```

### 2.2 定时聚合流

```
Hourly Aggregation Task (每小时第5分钟)
        │
        ├─► aggregateGroupStatusHourly()
        │   │
        │   ├─ Query consume_log [last_hour]
        │   ├─ Group by group_name
        │   ├─ Calculate metrics
        │   └─ INSERT group_status_history
        │
        └─► aggregateModelStatusHourly()
            │
            ├─ Query consume_log [last_hour]
            ├─ Group by model_name
            ├─ Calculate metrics
            └─ INSERT model_status_history
```

### 2.3 告警生成流

```
Alert Generation Task (每5分钟)
        │
        ├─► checkGroupHealthAlerts()
        │   │
        │   ├─ Query group_status
        │   ├─ Check if success_rate < threshold
        │   ├─ Verify no recent alert exists
        │   ├─ Generate recommendations
        │   │   ├─ Find better groups
        │   │   └─ Find alternative models
        │   └─ Create alert_notifications record
        │
        └─► checkModelAvailabilityAlerts()
            │
            ├─ Query model_status
            ├─ Check if available_channels decreased
            └─ Generate model_status_history
```

### 4.1 GroupStatusService

**service/group_status.go**:
```go
package service

import (
    "fmt"
    "sort"
    "time"
    "new-api-er/model"
    "new-api-er/common"
)

type GroupStatusService struct{}

var groupStatusService = &GroupStatusService{}

// 获取用户可见的分组状态
func (s *GroupStatusService) GetUserGroupStatus(userID int, userGroup string) ([]model.GroupStatus, error) {
    var groups []model.GroupStatus
    
    // 获取用户可用的分组
    usableGroups := GetUserUsableGroups(userGroup)
    
    // 查询这些分组的状态
    var groupNames []string
    for name := range usableGroups {
        groupNames = append(groupNames, name)
    }
    
    err := common.DB.Where("user_group IN ?", groupNames).
        Order("status ASC, updated_at DESC").
        Find(&groups).Error
    
    if err != nil {
        return nil, err
    }
    
    return groups, nil
}

// 计算指标（从consume_log）
func (s *GroupStatusService) CalculateGroupMetrics(groupName string, lastNHours int) (*model.GroupStatus, error) {
    type MetricResult struct {
        TotalCount      int
        SuccessCount    int
        AvgResponseTime float64
        P50ResponseTime int
        P95ResponseTime int
        ErrorCount      int
    }
    
    var result MetricResult
    
    // 查询consume_log中的数据
    query := common.DB.Table("consume_log").
        Where("group = ?", groupName).
        Where("created_at >= ?", time.Now().Unix() - int64(lastNHours*3600))
    
    // 统计指标
    err := query.Select(
        "COUNT(*) as total_count",
        "SUM(CASE WHEN error IS NULL THEN 1 ELSE 0 END) as success_count",
        "AVG(CAST(response_time AS DECIMAL)) as avg_response_time",
        "SUM(CASE WHEN error IS NOT NULL THEN 1 ELSE 0 END) as error_count",
    ).Scan(&result).Error
    
    if err != nil {
        return nil, err
    }
    
    // 查询百分位数（可能需要应用层计算）
    var responseTimes []int
    err = query.Select("response_time").Scan(&responseTimes).Error
    if err != nil {
        return nil, err
    }
    
    // 计算百分位数
    sort.Ints(responseTimes)
    p50, p95 := 0, 0
    if len(responseTimes) > 0 {
        p50 = percentile(responseTimes, 50)
        p95 = percentile(responseTimes, 95)
    }
    
    // 计算成功率
    successRate := 0.0
    if result.TotalCount > 0 {
        successRate = float64(result.SuccessCount) / float64(result.TotalCount) * 100
    }
    
    // 确定状态
    status := determineStatus(successRate, p95)
    
    groupStatus := &model.GroupStatus{
        UserGroup:       groupName,
        Status:          status,
        SuccessRate:     successRate,
        AvgResponseTime: int(result.AvgResponseTime),
        P50ResponseTime: p50,
        P95ResponseTime: p95,
        ErrorCount:      result.ErrorCount,
        LastTestTime:    time.Now().Unix(),
    }
    
    return groupStatus, nil
}

// 更新分组状态
func (s *GroupStatusService) UpdateGroupStatus(groupStatus *model.GroupStatus) error {
    groupStatus.UpdatedAt = time.Now().Unix()
    
    return common.DB.Save(groupStatus).Error
}

// 聚合小时数据并保存历史
func (s *GroupStatusService) AggregateHourlyHistory(groupName string) error {
    metrics, err := s.CalculateGroupMetrics(groupName, 1)
    if err != nil {
        return err
    }
    
    history := &model.GroupStatusHistory{
        UserGroup:        groupName,
        SnapshotHour:     (time.Now().Unix() / 3600) * 3600,
        SuccessRate:      metrics.SuccessRate,
        AvgResponseTime:  metrics.AvgResponseTime,
        ErrorCount:       metrics.ErrorCount,
        TotalRequests:    metrics.TotalChannels * 16, // 简化计算
        CreatedAt:        time.Now().Unix(),
    }
    
    return common.DB.Create(history).Error
}

// 辅助函数：百分位数计算
func percentile(data []int, p int) int {
    if len(data) == 0 {
        return 0
    }
    index := len(data) * p / 100
    if index >= len(data) {
        index = len(data) - 1
    }
    return data[index]
}

// 辅助函数：确定状态
func determineStatus(successRate float64, p95ResponseTime int) int {
    if successRate >= 99 && p95ResponseTime < 2000 {
        return 1 // 健康
    } else if successRate >= 95 && p95ResponseTime < 3000 {
        return 2 // 警告
    } else if successRate >= 90 {
        return 3 // 不健康
    }
    return 4 // 不可用
}

// 推荐更好的分组
func (s *GroupStatusService) RecommendBetterGroups(currentGroup string) ([]model.GroupStatus, error) {
    var betterGroups []model.GroupStatus
    
    // 查询所有分组状态
    var allGroups []model.GroupStatus
    err := common.DB.Order("status ASC, success_rate DESC").Find(&allGroups).Error
    if err != nil {
        return nil, err
    }
    
    // 找出当前分组
    currentStatus := &model.GroupStatus{}
    for _, g := range allGroups {
        if g.UserGroup == currentGroup {
            currentStatus = &g
            break
        }
    }
    
    // 找出更好的分组（状态更好或成功率更高）
    for _, g := range allGroups {
        if g.UserGroup == currentGroup {
            continue
        }
        
        if g.Status < currentStatus.Status ||
            (g.Status == currentStatus.Status && g.SuccessRate > currentStatus.SuccessRate) {
            betterGroups = append(betterGroups, g)
        }
    }
    
    return betterGroups, nil
}
```

**service/alert_service.go**:
```go
package service

import (
    "encoding/json"
    "fmt"
    "time"
    "new-api-er/model"
    "new-api-er/common"
    "new-api-er/dto"
)

type AlertService struct{}

var alertService = &AlertService{}

// 检查分组健康告警
func (s *AlertService) CheckGroupHealthAlerts() error {
    var groups []model.GroupStatus
    
    err := common.DB.Where("status > ?", 1).Find(&groups).Error
    if err != nil {
        return err
    }
    
    for _, group := range groups {
        // 检查是否已有未处理的相同告警
        var existingAlert model.AlertNotification
        result := common.DB.Where(
            "target_name = ? AND alert_type = ? AND status = ?",
            group.UserGroup,
            "group_unhealthy",
            "unread",
        ).Order("created_at DESC").Limit(1).Find(&existingAlert)
        
        if result.RowsAffected > 0 {
            // 检查是否超过6小时（防止告警风暴）
            if time.Now().Unix() - existingAlert.CreatedAt < 6*3600 {
                continue
            }
        }
        
        // 生成新告警
        alert := &model.AlertNotification{
            UserID:     0, // TODO: 获取该分组的所有用户
            Title:      fmt.Sprintf("分组 '%s' 健康度下降", group.GroupName),
            Message:    fmt.Sprintf("成功率 %.1f%% (阈值: 95%%), 建议检查或切换分组", group.SuccessRate),
            AlertType:  "group_unhealthy",
            TargetName: group.UserGroup,
            Status:     "unread",
            CreatedAt:  time.Now().Unix(),
        }
        
        // 生成推荐
        recommendations := s.generateGroupRecommendations(&group)
        recBytes, _ := json.Marshal(recommendations)
        alert.Recommendation = string(recBytes)
        
        // 保存告警
        if err := common.DB.Create(alert).Error; err != nil {
            common.LogWarnf("Failed to create alert for group %s: %v", group.UserGroup, err)
        }
    }
    
    return nil
}

// 生成分组推荐
func (s *AlertService) generateGroupRecommendations(group *model.GroupStatus) []dto.Recommendation {
    var recs []dto.Recommendation
    
    // 找出更好的分组
    betterGroups, _ := groupStatusService.RecommendBetterGroups(group.UserGroup)
    if len(betterGroups) > 0 {
        bestGroup := betterGroups[0]
        recs = append(recs, dto.Recommendation{
            Action: "switch_group",
            Target: bestGroup.UserGroup,
            Reason: fmt.Sprintf(
                "推荐分组 '%s' 成功率 %.1f%%, 响应时间 %dms",
                bestGroup.GroupName,
                bestGroup.SuccessRate,
                bestGroup.AvgResponseTime,
            ),
        })
    }
    
    // 建议检查该分组的禁用渠道
    if group.DisabledChannels > 0 {
        recs = append(recs, dto.Recommendation{
            Action: "check_channels",
            Reason: fmt.Sprintf("该分组有 %d 个禁用渠道，建议检查配置", group.DisabledChannels),
        })
    }
    
    return recs
}

// 标记告警为已读
func (s *AlertService) MarkAlertAsRead(alertID int64, actionTaken string) error {
    update := map[string]interface{}{
        "status":       "read",
        "read_at":      time.Now().Unix(),
        "action_taken": actionTaken,
    }
    
    return common.DB.Model(&model.AlertNotification{}).
        Where("id = ?", alertID).
        Updates(update).Error
}

// 获取用户的未读告警
func (s *AlertService) GetUserUnreadAlerts(userID int, limit int) ([]model.AlertNotification, error) {
    var alerts []model.AlertNotification
    
    err := common.DB.Where("user_id = ? AND status = ?", userID, "unread").
        Order("created_at DESC").
        Limit(limit).
        Find(&alerts).Error
    
    return alerts, err
}
```

```go
    groupName := c.Param("group")
    daysStr := c.DefaultQuery("days", "7")

        "data": gin.H{

            "history":    history,
        },
    })
```

```go
package controller

import (
    "net/http"
    "strconv"
    "github.com/gin-gonic/gin"
    "new-api-er/service"
)

// GET /api/alerts?status=unread&limit=20
func GetAlerts(c *gin.Context) {
    userID := c.GetInt("user_id")
    status := c.DefaultQuery("status", "unread")
    limitStr := c.DefaultQuery("limit", "20")
    limit, _ := strconv.Atoi(limitStr)
    
    var alerts []model.AlertNotification
    
    query := common.DB.Where("user_id = ?", userID)
    if status != "all" {
        query = query.Where("status = ?", status)
    }
    
    err := query.Order("created_at DESC").
        Limit(limit).
        Find(&alerts).Error
    
    if err != nil {
        c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
        return
    }
    
    // 统计未读数
    var unreadCount int64
    common.DB.Where("user_id = ? AND status = ?", userID, "unread").
        Model(&model.AlertNotification{}).Count(&unreadCount)
    
    c.JSON(http.StatusOK, gin.H{
        "success":       true,
        "unread_count":  unreadCount,
        "total_count":   len(alerts),
        "alerts":        alerts,
    })
}

// POST /api/alerts/:alert_id/mark_read
func MarkAlertAsRead(c *gin.Context) {
    alertIDStr := c.Param("alert_id")
    alertID, _ := strconv.ParseInt(alertIDStr, 10, 64)
    
    var req dto.MarkAlertReadRequest
    c.BindJSON(&req)
    
    err := service.alertService.MarkAlertAsRead(alertID, req.ActionTaken)
    if err != nil {
        c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
        return
    }
    
    c.JSON(http.StatusOK, gin.H{"success": true})
}
```

```go
func StartGroupStatusAggregationTask() {
    if !common.IsMasterNode {
        return
    }
    
    // 等待到下一个整点+5分钟
    now := time.Now()
    nextHour := now.Add(time.Hour)
    nextRun := time.Date(
        nextHour.Year(), nextHour.Month(), nextHour.Day(),
        nextHour.Hour(), 5, 0, 0, nextHour.Location(),
    )
    duration := nextRun.Sub(now)
    
    time.Sleep(duration)
    
    ticker := time.NewTicker(1 * time.Hour)
    defer ticker.Stop()
    
    for range ticker.C {
        // 查询所有分组
        var groups []model.GroupStatus
        if err := common.DB.Find(&groups).Error; err != nil {
            common.LogErrorf("Failed to query groups: %v", err)
            continue
        }
        
        // 对每个分组进行聚合
        for _, group := range groups {
            if err := service.groupStatusService.AggregateHourlyHistory(group.UserGroup); err != nil {
                common.LogErrorf("Failed to aggregate history for group %s: %v", group.UserGroup, err)
            }
        }
        
        // 清理旧数据（>90天）
        cutoffTime := time.Now().Unix() - 90*24*3600
        common.DB.Where("created_at < ?", cutoffTime).
            Delete(&model.GroupStatusHistory{})
    }
}

            common.LogErrorf("Failed to check model availability alerts: %v", err)
```

## 9. 性能优化建议

### 9.1 缓存策略

```go
// 缓存分组状态（5分钟TTL）
const GroupStatusCacheTTL = 5 * time.Minute

var groupStatusCache map[string]*model.GroupStatus
var groupStatusCacheMutex sync.RWMutex
var groupStatusCacheTime time.Time

func GetGroupStatusCached(groupName string) (*model.GroupStatus, error) {
    groupStatusCacheMutex.RLock()
    if time.Since(groupStatusCacheTime) < GroupStatusCacheTTL {
        defer groupStatusCacheMutex.RUnlock()
        return groupStatusCache[groupName], nil
    }
    groupStatusCacheMutex.RUnlock()
    
    // 缓存过期，重新查询
    return GetGroupStatusFromDB(groupName)
}
```

### 9.2 批量操作优化

```go
// 批量查询分组状态
func GetAllGroupStatusBatch() (map[string]*model.GroupStatus, error) {
    var groups []model.GroupStatus
    err := common.DB.Find(&groups).Error
    
    result := make(map[string]*model.GroupStatus, len(groups))
    for i := range groups {
        result[groups[i].UserGroup] = &groups[i]
    }
    
    return result, err
}
```
